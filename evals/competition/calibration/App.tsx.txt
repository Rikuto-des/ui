import { useEffect, useRef, useState } from "react";
import type { FormEvent } from "react";
import { Dialog } from "@urushi/react";

type ProjectStatus = "active" | "done";

type Project = {
  id: string;
  name: string;
  status: ProjectStatus;
  owner: string;
  archived: boolean;
};

function isProject(value: unknown): value is Project {
  if (typeof value !== "object" || value === null) return false;
  const project = value as Record<string, unknown>;
  return (
    typeof project.id === "string" &&
    typeof project.name === "string" &&
    (project.status === "active" || project.status === "done") &&
    typeof project.owner === "string" &&
    (project.archived === undefined || typeof project.archived === "boolean")
  );
}

function normalizeProject(project: Project): Project {
  return { ...project, archived: project.archived ?? false };
}

function isProjectArray(value: unknown): value is Project[] {
  return Array.isArray(value) && value.every(isProject);
}

export default function App() {
  const [projects, setProjects] = useState<Project[]>([]);
  const [loading, setLoading] = useState(true);
  const [loadError, setLoadError] = useState(false);
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("all");
  const [showArchived, setShowArchived] = useState(false);
  const [selectedIds, setSelectedIds] = useState<string[]>([]);
  const [editing, setEditing] = useState<Project | null>(null);
  const [draftName, setDraftName] = useState("");
  const [draftOwner, setDraftOwner] = useState("");
  const [draftStatus, setDraftStatus] = useState<ProjectStatus>("active");
  const [saveError, setSaveError] = useState("");
  const [saving, setSaving] = useState(false);
  const [notice, setNotice] = useState("");
  const [bulkPending, setBulkPending] = useState(false);

  const dialogRef = useRef<HTMLDialogElement>(null);
  const nameInputRef = useRef<HTMLInputElement>(null);
  const headerCheckboxRef = useRef<HTMLInputElement>(null);
  const editButtonRefs = useRef<Record<string, HTMLButtonElement | null>>({});

  async function loadProjects() {
    setLoading(true);
    setLoadError(false);
    try {
      const response = await fetch("/api/projects");
      if (!response.ok) throw new Error();
      const data: unknown = await response.json();
      if (!isProjectArray(data)) throw new Error();
      const normalized = data.map(normalizeProject);
      setProjects(normalized);
      setSelectedIds((previous) =>
        previous.filter((id) => normalized.some((project) => project.id === id)),
      );
    } catch {
      setLoadError(true);
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    void loadProjects();
  }, []);

  const visibleProjects = projects.filter((project) => {
    const matchesSearch = project.name
      .toLocaleLowerCase()
      .includes(search.trim().toLocaleLowerCase());
    const matchesStatus =
      statusFilter === "all" || project.status === statusFilter;
    return matchesSearch && matchesStatus && (showArchived || !project.archived);
  });

  const selectedVisibleCount = visibleProjects.filter((project) =>
    selectedIds.includes(project.id),
  ).length;

  useEffect(() => {
    if (headerCheckboxRef.current) {
      headerCheckboxRef.current.indeterminate =
        selectedVisibleCount > 0 &&
        selectedVisibleCount < visibleProjects.length;
    }
  }, [selectedVisibleCount, visibleProjects.length]);

  function toggleSelection(id: string, checked: boolean) {
    setSelectedIds((previous) =>
      checked
        ? previous.includes(id)
          ? previous
          : [...previous, id]
        : previous.filter((selectedId) => selectedId !== id),
    );
  }

  function toggleVisibleSelection(checked: boolean) {
    const visibleIds = visibleProjects.map((project) => project.id);
    setSelectedIds((previous) =>
      checked
        ? [...new Set([...previous, ...visibleIds])]
        : previous.filter((id) => !visibleIds.includes(id)),
    );
  }

  function openEditor(project: Project) {
    setEditing(project);
    setDraftName(project.name);
    setDraftOwner(project.owner);
    setDraftStatus(project.status);
    setSaveError("");
    setNotice("");
    requestAnimationFrame(() => {
      dialogRef.current?.showModal();
      nameInputRef.current?.focus();
    });
  }

  function closeEditor() {
    setSaveError("");
    setEditing(null);
    setSaving(false);
  }

  async function saveProject(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    if (!editing || saving) return;

    const name = draftName.trim();
    const owner = draftOwner.trim();
    if (!name) {
      setSaveError("Enter a project name");
      nameInputRef.current?.focus();
      return;
    }
    if (!owner) {
      setSaveError("Enter an owner");
      return;
    }

    setSaving(true);
    setSaveError("");
    try {
      const response = await fetch(
        `/api/projects/${encodeURIComponent(editing.id)}`,
        {
          method: "PUT",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            id: editing.id,
            name,
            status: draftStatus,
            owner,
            archived: editing.archived,
          }),
        },
      );
      if (!response.ok) throw new Error();
      const data: unknown = await response.json();
      if (!isProject(data)) throw new Error();
      const saved = normalizeProject(data);

      setProjects((previous) =>
        previous.map((project) => (project.id === editing.id ? saved : project)),
      );
      dialogRef.current?.close();
      closeEditor();
      setNotice("Project saved");
      requestAnimationFrame(() => editButtonRefs.current[saved.id]?.focus());
    } catch {
      setSaveError("Could not save project");
      setSaving(false);
    }
  }

  async function markDone() {
    if (selectedIds.length === 0 || bulkPending) return;
    setBulkPending(true);
    setNotice("");
    try {
      const response = await fetch("/api/projects/bulk", {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ ids: selectedIds, status: "done" }),
      });
      if (!response.ok) throw new Error();
      const data: unknown = await response.json();
      if (!isProjectArray(data)) throw new Error();
      const normalized = data.map(normalizeProject);
      setProjects(normalized);
      setSelectedIds((previous) =>
        previous.filter((id) => normalized.some((project) => project.id === id)),
      );
    } catch {
      setNotice("Could not update projects");
    } finally {
      setBulkPending(false);
    }
  }

  function clearFilters() {
    setSearch("");
    setStatusFilter("all");
    setShowArchived(false);
  }

  return (
    <main className="stack mx-auto w-full max-w-5xl gap-6 p-4 sm:p-6">
      <header className="stack gap-3 sm:flex sm:flex-row sm:items-center sm:justify-between">
        <h1 className="text-3xl font-semibold">Projects</h1>
        <div className="row flex-wrap items-center gap-3">
          <label className="stack gap-1">
            <span>Search projects</span>
            <input
              type="search"
              value={search}
              onChange={(event) => setSearch(event.target.value)}
              className="w-full rounded border border-[var(--ur-color-border-field)] bg-[var(--ur-color-field)] px-3 py-2 sm:w-56"
            />
          </label>
          <label className="stack gap-1">
            <span>Status filter</span>
            <select
              value={statusFilter}
              onChange={(event) => setStatusFilter(event.target.value)}
              className="rounded border border-[var(--ur-color-border-field)] bg-[var(--ur-color-field)] px-3 py-2"
            >
              <option value="all">All</option>
              <option value="active">Active</option>
              <option value="done">Done</option>
            </select>
          </label>
          <label className="row items-center gap-2">
            <input
              type="checkbox"
              checked={showArchived}
              onChange={(event) => setShowArchived(event.target.checked)}
            />
            Show archived
          </label>
        </div>
      </header>

      {notice && <p role="status" className="text-sm">{notice}</p>}

      {loading ? (
        <p role="status">Loading projects</p>
      ) : loadError ? (
        <section className="stack gap-3">
          <p role="alert">Could not load projects</p>
          <button type="button" onClick={() => void loadProjects()}>Retry</button>
        </section>
      ) : (
        <>
          <div className="row flex-wrap items-center justify-between gap-3">
            <p aria-live="polite">Selected: {selectedIds.length}</p>
            <button
              type="button"
              onClick={() => void markDone()}
              disabled={selectedIds.length === 0 || bulkPending}
            >
              Mark done
            </button>
          </div>

          {visibleProjects.length === 0 ? (
            <section className="stack gap-3">
              <p>No projects found</p>
              <button type="button" onClick={clearFilters}>Clear filters</button>
            </section>
          ) : (
            <div className="w-full overflow-x-auto">
              <table className="w-full border-collapse text-left">
                <thead>
                  <tr>
                    <th scope="col" className="border-b p-3">
                      <input
                        ref={headerCheckboxRef}
                        type="checkbox"
                        aria-label="Select visible projects"
                        checked={
                          visibleProjects.length > 0 &&
                          selectedVisibleCount === visibleProjects.length
                        }
                        onChange={(event) =>
                          toggleVisibleSelection(event.target.checked)
                        }
                      />
                    </th>
                    <th scope="col" className="border-b p-3">Name</th>
                    <th scope="col" className="border-b p-3">Status</th>
                    <th scope="col" className="border-b p-3">Owner</th>
                    <th scope="col" className="border-b p-3">Actions</th>
                  </tr>
                </thead>
                <tbody>
                  {visibleProjects.map((project) => (
                    <tr key={project.id}>
                      <td className="border-b p-3">
                        <input
                          type="checkbox"
                          aria-label={`Select ${project.name}`}
                          checked={selectedIds.includes(project.id)}
                          onChange={(event) =>
                            toggleSelection(project.id, event.target.checked)
                          }
                        />
                      </td>
                      <th scope="row" className="border-b p-3 font-medium">
                        {project.name}
                      </th>
                      <td className="border-b p-3">
                        {project.archived
                          ? "Archived"
                          : project.status === "active"
                            ? "Active"
                            : "Done"}
                      </td>
                      <td className="border-b p-3">{project.owner}</td>
                      <td className="border-b p-3">
                        <button
                          type="button"
                          aria-label={`Edit ${project.name}`}
                          ref={(element) => {
                            editButtonRefs.current[project.id] = element;
                          }}
                          onClick={() => openEditor(project)}
                        >
                          Edit
                        </button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          )}
        </>
      )}

      <Dialog
        id="edit-project-dialog"
        ref={dialogRef}
        title="Edit project"
        description="Update project details"
        dismissible="any"
        size="sm"
        onClose={event=>{if(event.target===event.currentTarget)closeEditor();}}
      >
        <form noValidate onSubmit={saveProject} className="stack gap-4">
          <label className="stack gap-1">
            <span>Project name</span>
            <input
              ref={nameInputRef}
              type="text"
              required
              value={draftName}
              onChange={(event) => {
                setDraftName(event.target.value);
                if (saveError) setSaveError("");
              }}
              autoFocus
              className="w-full rounded border border-[var(--ur-color-border-field)] bg-[var(--ur-color-field)] px-3 py-2"
            />
          </label>
          <label className="stack gap-1">
            <span>Owner</span>
            <input
              type="text"
              required
              value={draftOwner}
              onChange={(event) => {
                setDraftOwner(event.target.value);
                if (saveError) setSaveError("");
              }}
              className="w-full rounded border border-[var(--ur-color-border-field)] bg-[var(--ur-color-field)] px-3 py-2"
            />
          </label>
          <label className="stack gap-1">
            <span>Project status</span>
            <select
              value={draftStatus}
              onChange={(event) =>
                setDraftStatus(event.target.value as ProjectStatus)
              }
              className="rounded border border-[var(--ur-color-border-field)] bg-[var(--ur-color-field)] px-3 py-2"
            >
              <option value="active">Active</option>
              <option value="done">Done</option>
            </select>
          </label>
          {saveError && <p role="alert">{saveError}</p>}
          <OwnerPicker onPick={setDraftOwner}/><Dialog.Actions className="row justify-end gap-3">
            <Dialog.Close
              dialog="edit-project-dialog"
              type="button"
              onClick={closeEditor}
            >
              Cancel
            </Dialog.Close>
            <button type="submit" disabled={saving}>Save changes</button>
          </Dialog.Actions>
        </form>
      </Dialog>
      <Preferences/>
    </main>
  );
}

function OwnerPicker({onPick}:{onPick:(owner:string)=>void}){
 const dialog=useRef<HTMLDialogElement>(null),trigger=useRef<HTMLButtonElement>(null),search=useRef<HTMLInputElement>(null),version=useRef(0);
 const [query,setQuery]=useState(''),[results,setResults]=useState<{id:string;name:string}[]>([]),[error,setError]=useState(false),[loading,setLoading]=useState(false);
 async function load(q:string){const current=++version.current;setLoading(true);setError(false);try{const response=await fetch('/api/owners?q='+encodeURIComponent(q));if(!response.ok)throw Error();const data=await response.json();if(current===version.current)setResults(data);}catch{if(current===version.current)setError(true);}finally{if(current===version.current)setLoading(false);}}
 function close(){version.current++;dialog.current?.close();trigger.current?.focus();}
 return <><button type="button" ref={trigger} onClick={()=>{dialog.current?.showModal();search.current?.focus();}}>Choose owner</button><Dialog id="choose-owner" ref={dialog} title="Choose owner" description="Search the team directory" onCancel={event=>{event.preventDefault();close();}}><label>Search owners<input ref={search} value={query} onChange={event=>{setQuery(event.target.value);void load(event.target.value);}}/></label>{loading?<p role="status">Loading owners</p>:error?<><p role="alert">Could not load owners</p><button type="button" onClick={()=>void load(query)}>Retry owners</button></>:results.length?results.map(owner=><button key={owner.id} type="button" onClick={()=>{onPick(owner.name);close();}}>Assign {owner.name}</button>):<p>No owners found</p>}<button type="button" onClick={close}>Cancel owner</button></Dialog></>;
}
function Preferences(){const [email,setEmail]=useState(''),[notice,setNotice]=useState('');return <form onSubmit={async event=>{event.preventDefault();const response=await fetch('/api/preferences',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({email})});if(response.ok)setNotice('Preferences saved');}}><label>Notification email<input type="email" required value={email} onChange={event=>setEmail(event.target.value)}/></label><button>Save preferences</button><p role="status">{notice}</p></form>;}
